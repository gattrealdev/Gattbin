case 'reactwa': {
  const input = body.slice(8).trim();
  const [linkPart, emojiPart, countPart] = input.split('|');
  
  if (!linkPart || !emojiPart || !countPart) {
    return sock.sendMessage(from, {
      text: '❌ Format salah!\n\nContoh penggunaan:\n.reactwa https://whatsapp.com/channel/0029VbBfipy2975GOjz9uf1Y/135|😀,😍,👍|10'
    }, { quoted: m });
  }

  try {
    const channelRegex = /channel\/([A-Za-z0-9]+)\/(\d+)/;
    const match = linkPart.match(channelRegex);
    
    if (!match) {
      return sock.sendMessage(from, {
        text: '❌ Link channel tidak valid!\nPastikan link mengandung ID channel dan message ID.\nContoh: https://whatsapp.com/channel/0029VbBfipy2975GOjz9uf1Y/135'
      }, { quoted: m });
    }

    const channelId = match[1];
    const messageId = match[2];
    const emojis = emojiPart.split(',').map(e => e.trim()).filter(e => e);
    const count = parseInt(countPart);
    
    if (emojis.length === 0) {
      return sock.sendMessage(from, {
        text: '❌ Minimal 1 emoji diperlukan!'
      }, { quoted: m });
    }
    
    if (isNaN(count) || count < 1 || count > 15000) {
      return sock.sendMessage(from, {
        text: '❌ Jumlah harus antara 1-15000!'
      }, { quoted: m });
    }

    const remoteJid = `${channelId}@newsletter`;
    const messageKey = {
      remoteJid: remoteJid,
      id: messageId,
      fromMe: false,
      participant: undefined
    };

    let successCount = 0;
    let failedCount = 0;
    
    const progressMsg = await sock.sendMessage(from, {
      text: `⏳ Memulai pengiriman reaction...\n📊 Target: ${count} reactions\n🎯 Emoji: ${emojis.join(', ')}`
    });

    const delay = ms => new Promise(resolve => setTimeout(resolve, ms));

    for (let i = 0; i < count; i++) {
      try {
        const emoji = emojis[i % emojis.length];
        
        await sock.sendMessage(remoteJid, {
          react: {
            text: emoji,
            key: messageKey
          }
        });
        
        successCount++;
        
        if ((i + 1) % 5 === 0 || i === count - 1) {
          await sock.sendMessage(from, {
            text: `📊 Progress: ${i + 1}/${count} reactions\n✅ Berhasil: ${successCount}\n❌ Gagal: ${failedCount}`
          }, { quoted: progressMsg });
        }
        
        await delay(500 + Math.random() * 500);
        
      } catch (error) {
        failedCount++;
        console.error(`Error reaction ${i + 1}:`, error);
      }
    }


    await sock.sendMessage(from, {
      text: `✅ Selesai!\n\n📊 Hasil akhir:\n├ Total dikirim: ${count}\n├ Berhasil: ${successCount}\n└ Gagal: ${failedCount}\n\n🎯 Emoji yang digunakan: ${emojis.join(', ')}`
    }, { quoted: progressMsg });

  } catch (error) {
    console.error(error);
    await sock.sendMessage(from, {
      text: `❌ Terjadi kesalahan:\n${error.message}`
    }, { quoted: m });
  }
  break;
}
